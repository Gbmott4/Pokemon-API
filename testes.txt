POST
curl.exe -X POST "http://localhost:3333/api/v1/pokemons" -H "Content-Type: application/json" -d '{\"id\":\"25\",\"name\":\"Pikachu\",\"type\":\"Electric\",\"hp\":35,\"attack\":55,\"defense\":40}'

GET — Listar todos
curl.exe "http://localhost:3333/api/v1/pokemons"

GET — Filtrar por tipo
curl.exe "http://localhost:3333/api/v1/pokemons?type=Electric"

GET — Buscar pelo ID
curl.exe "http://localhost:3333/api/v1/pokemons/25"

PUT — Atualizar Pikachu
curl.exe --% -X PUT http://localhost:3333/api/v1/pokemons/25 -H "Content-Type: application/json" -d "{\"name\":\"Pikachu Fortalecido\",\"type\":\"Electric\",\"hp\":50,\"attack\":65,\"defense\":45}"

DELETE — Remover Pikachu
curl.exe -X DELETE "http://localhost:3333/api/v1/pokemons/25"
